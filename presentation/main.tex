\section{Introduction}

\begin{frame}{Introduction}
\begin{block}{Problem}
Given $f,g\in R[x]$ with $\deg(f)=p>q=\deg(g)$, compute
\[
  \gcd(f,g), \qquad \Res(f,g), \qquad \bigl(\sResP_i(f,g)\bigr)_{0\le i\le q}.
\]
\end{block}
\pause
\begin{block}{Main idea}
Use subresultants and specialization to replace expensive symbolic computations by many cheaper univariate computations.
\end{block}
\pause
\begin{block}{Background}
Univariate and multivariate polynomials, GCD, resultants, subresultants and subresultant polynomials.
\end{block}
\end{frame}

\begin{frame}{Euclidean Algorithm: Correct but Costly}
\footnotesize
\begin{columns}
  \begin{column}{0.45\textwidth}
    Euclidean GCD: \\
    \vspace{-2em}
    \[
      r_0=f, \qquad r_1=g, \qquad r_{i+1}=r_{i-1}\bmod r_i.
    \]
    \vspace{-2em}
    \pause
    \begin{alertblock}{Issue}
    Over rings such as $\Z$, coefficient sizes may grow quickly during division.
    \end{alertblock}
    \pause
    \vspace{-1em}
    \[
    \begin{aligned}
    f&=x^8+x^6-3x^4-3x^3+8x^2+2x-5,\\
    g&=3x^6+5x^4-4x^2-9x+21.
    \end{aligned}
    \]
    \vspace{1em}
    \[
    \begin{aligned}
    r_2 &= -\frac{5}{9}x^4+\frac{1}{9}x^2-\frac{1}{3}, &
    r_3 &= -\frac{117}{25}x^2-9x+\frac{441}{25},\\
    r_4 &= \frac{233150}{19773}x-\frac{102500}{6591}, &
    r_5 &= -\frac{1288744821}{543589225}.
    \end{aligned}
    \]
    \vspace{1em}
    \[
      \gcd(f,g)=1.
    \]
  \end{column}
  \pause
  \begin{column}{0.45\textwidth}
    Using Subresultants:
    \begin{block}{Key property}
    The subresultant subsequence of polynomials is proportional to the Euclidean remainder sequence, and GCD is the final non-zero subresultant polynomial.
    \end{block}
    \pause
    For the same $f$ and $g$:
    \[
    \begin{aligned}
    \sResP_5(f,g) &= 15x^4-3x^2+9, \\
    \sResP_3(f,g) &= 65x^2+125x-245,\\
    \sResP_2(f,g) &= 9326x-12300, \\
    \sResP_0(f,g) &= 260708.
    \end{aligned}
    \]
    \vspace{0.5em}
    \[
      \gcd(f,g)=1.
    \]
  \end{column}
\end{columns}
\end{frame}

\section{State of the Art and Contribution}

\begin{frame}{State of the Art and Contribution}
  \begin{columns}
    \begin{column}{0.45\textwidth}
      State of the art:
      \begin{itemize}
        \item 1750, Subresultants were already known to Euler.
        \pause
        \item 1967, Collins used them heavily in computational algebra.
        \pause
        \item 2000, Lazard and Ducos improved the subresultant algorithm.
        \pause
        \item today, FLINT and Maple use subresultant-based methods for resultants.
        \item today, Wolfram exposes \texttt{SubresultantPolynomials} directly.
      \end{itemize}
    \end{column}
    \pause
    \begin{column}{0.45\textwidth}
      \begin{block}{This work}
      Design and implement (on FLINT) a fast algorithm for subresultants over $\Q$, $\Z$, and $\Fp$.
      \end{block}
      \pause
      \begin{block}{Technique}
      Use evoluation and interpolation, generalizing Collins's resultant algorithm.
      \end{block}
      \pause
      \begin{block}{Observed performance}
      Speedups up to $7\times$ on benchmarks with total degree of bitsize $6$ and coefficient bit size $50$, both on FMPZ and FMPQ.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Subresultant Polynomials}
\footnotesize
\begin{columns}
\begin{column}{0.45\textwidth}
For $0\le i\le q$ and $0\le j\le i$, let $M_{i,j}(f,g)$ be the corresponding Sylvester-Habicht minor.

\begin{block}{Definition}
\[
  \sResP_i(f,g)=\sum_{j=0}^i \det(M_{i,j}(f,g))x^j.
\]
\end{block}

\begin{block}{Non-defective case}
\[
  \deg\bigl(\sResP_i(f,g)\bigr)=i.
\]
\end{block}
\end{column}
\pause
\begin{column}{0.45\textwidth}
\begin{itemize}
  \item Non-zero subresultants encode Euclidean remainders.
  \item Defective subresultants reveal degree drops.
  \item Zero subresultants indicate gaps in the sequence.
  \item Last non-zero subresultant polynomial equals the GCD.
\end{itemize}

\vspace{0.8em}
\begin{block}{Consequence}
The GCD and resultant can be computed without division in the fraction field.
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Proposed Algorithm}

\begin{frame}{Multivariate Setting}
  \begin{columns}
    \begin{column}{0.45 \textwidth}
      Let
      \[
        f,g\in \Z[x_1,\dots,x_n].
      \]

      View them as univariate polynomials in $x_1$:
      \[
        f,g\in R[x_1], \qquad R=\Z[x_2,
      \dots,x_n].
      \]

      \begin{block}{Goal}
      Compute
      \[
        \Res_{x_1}(f,g), \qquad \sResP_i(f,g).
      \]
      \end{block}
    \end{column}
    \pause
    \begin{column}{0.45 \textwidth}
      Symbolic computation over
      \[
        R=\Z[x_2,\dots,x_n]
      \]
      can be expensive.

      \vspace{0.5em}
      Instead, choose evaluation points
      \[
        \eta\in\Z^{n-1}
      \]
      and compute in
      \[
        \Z[x_1].
      \]
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Backbone of the Approach}
\footnotesize
\begin{columns}
  \begin{column}{0.45 \textwidth}
    \textbf{Specialization of the Resultant}\\
    Let
    \[
      h=\lc_{x_1}(f)\lc_{x_1}(g).
    \]


    \begin{block}{Valid evaluation point}
    If $h(\eta)\ne 0$, then
    \[
      \Res_{x_1}(f(\eta),g(\eta))
      =
      \Res_{x_1}(f,g)(\eta).
    \]
    \end{block}
    \pause
    \begin{alertblock}{Bad points}
    Degree drops after evaluation can break the equality.
    \end{alertblock}
  \end{column}
  \pause
  \begin{column}{0.45 \textwidth}
    \textbf{Degree Bound}\\
    Let
    \[
    D=\deg_{x_1}(f)\deg_{x_2,\dots,x_n}(g)
    +
    \deg_{x_1}(g)\deg_{x_2,\dots,x_n}(f).
    \]
    \pause
    \begin{block}{Bound}
    \[
      \deg\bigl(\Res_{x_1}(f,g)\bigr)\le D.
    \]
    \end{block}
    \pause
    \begin{block}{Interpolation cost}
    For $n=2$, at most $D+1$ evaluation points are needed.
    \end{block}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Proposed Bivariate Resultant Algorithm}
\begin{columns}
  \begin{column}{0.45 \textwidth}
    For $f,g\in\Z[x_1,x_2]$:
    \[
      N=\deg_{x_1}(f)\deg_{x_2}(g)
      +
      \deg_{x_1}(g)\deg_{x_2}(f).
    \]

    \begin{enumerate}
      \item Choose $N+1$ good evaluation points.
      \item Compute
      \[
        y_i=\Res(f(x_i),g(x_i)).
      \]
      \item Interpolate
      \[
        R=\operatorname{interpolate}(x_i,y_i,N+1).
      \]
    \end{enumerate}
  \end{column}
  \pause
  \begin{column}{0.45 \textwidth}
    Implemented strategies:
    \begin{itemize}
      \item Random Points
      \item Random Positive Points
      \item Small Random Points
      \item Small Random Positive Points
      \item Ordered Points
      \item Ordered Positive Points
    \end{itemize}

    \vspace{0.5em}
    Small and ordered points usually reduce interpolation cost.
  \end{column}
\end{columns}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
\footnotesize
\begin{columns}
\begin{column}{0.45\textwidth}
  \textbf{Summary of Results}
  \begin{itemize}
    \item Subresultants control coefficient growth in GCD/resultant computations.
    \item Specialization is valid away from leading-coefficient zeros.
    \item Interpolation recovers multivariate resultants and subresultants.
    \item Evaluation point choice is critical for performance.
    \item Ordered or small-bit-size points work best in the benchmarks.
    \item Our implementation improves FLINT's performance, especially for polynomials with high degree and length.
  \end{itemize}
\end{column}
\pause
\begin{column}{0.45\textwidth}
  \textbf{Next Steps}
  \begin{itemize}
    \item Benchmark subresultant implementation for bivariate polynomials.
    \item Extend implementation from resultants / subresultants to subresultant polynomials.
    \item Extend implementation to higher dimension multivariate polynomials (note that number of points for interpolation increases - $(D + 1)^{n-1}$).
    \item Optimize, test, and benchmark for higher degree and length polynomials.
  \end{itemize}
\end{column}
\end{columns}
\end{frame}
