\section{Introduction}

\begin{frame}{Goal}
\begin{block}{Problem}
Given $f,g\in R[x]$ with $\deg(f)=p>q=\deg(g)$, compute
\[
  \gcd(f,g), \qquad \Res(f,g), \qquad \bigl(\sResP_i(f,g)\bigr)_{0\le i\le q}.
\]
\end{block}
\begin{block}{Main idea}
Use subresultants and specialization to replace expensive symbolic computations by many cheaper univariate computations.
\end{block}
\end{frame}

\begin{frame}{Euclidean Algorithm: Correct but Costly}
\[
  r_0=f, \qquad r_1=g, \qquad r_{i+1}=r_{i-1}\bmod r_i.
\]

\vspace{0.4em}
The last non-zero remainder is $\gcd(f,g)$.

\vspace{0.8em}
\begin{alertblock}{Issue}
Over rings such as $\Z$, coefficient sizes may grow quickly during division.
\end{alertblock}
\end{frame}

\begin{frame}{Example: Coefficient Growth}
\small
\[
\begin{aligned}
 f&=x^8+x^6-3x^4-3x^3+8x^2+2x-5,\\
 g&=3x^6+5x^4-4x^2-9x+21.
\end{aligned}
\]

\[
\begin{aligned}
 r_2 &= -\frac{5}{9}x^4+\frac{1}{9}x^2-\frac{1}{3}, &
 r_3 &= -\frac{117}{25}x^2-9x+\frac{441}{25},\\
 r_4 &= \frac{233150}{19773}x-\frac{102500}{6591}, &
 r_5 &= -\frac{1288744821}{543589225}.
\end{aligned}
\]

\[
  \gcd(f,g)=1.
\]
\end{frame}

\begin{frame}{Subresultants}
\begin{block}{Key property}
An appropriate subsequence of
\[
  \bigl(\sResP_i(f,g)\bigr)_{0\le i\le q}
\]
is proportional to the Euclidean remainder sequence.
\end{block}

\vspace{0.5em}
They are defined by determinants, so coefficients remain in the base ring.
\end{frame}

\begin{frame}{Example: Subresultants}
For the same $f$ and $g$:
\[
\begin{aligned}
\sResP_5(f,g) &= 15x^4-3x^2+9, &
\sResP_3(f,g) &= 65x^2+125x-245,\\
\sResP_2(f,g) &= 9326x-12300, &
\sResP_0(f,g) &= 260708.
\end{aligned}
\]

\vspace{0.5em}
\[
  \gcd(f,g)=1.
\]
\end{frame}

\begin{frame}{State of the Art}
\begin{itemize}
  \item Subresultants were already known to Euler.
  \item Collins introduced their use in computer algebra.
  \item Lazard and Ducos improved the subresultant algorithm.
  \item FLINT and Maple use subresultant-based methods for resultants.
  \item Wolfram exposes \texttt{SubresultantPolynomials} directly.
\end{itemize}
\end{frame}

\begin{frame}{Contribution}
\begin{block}{This work}
Design a fast algorithm for subresultants over
\[
  \Q, \qquad \Z, \qquad \Fp.
\]
\end{block}

\begin{block}{Technique}
Use specialization and interpolation, generalizing Collins's resultant algorithm.
\end{block}

\begin{block}{Observed performance}
Speedups up to $20\times$ on benchmarks with total degree $30$ and coefficient bit size $100$.
\end{block}
\end{frame}

\section{Resultants}

\begin{frame}{Sylvester Matrix}
Let
\[
 f=a_p x^p+\cdots+a_0, \qquad g=b_qx^q+\cdots+b_0.
\]

The Sylvester matrix represents
\[
\phi:R[x]_{<q}\oplus R[x]_{<p}\to R[x]_{<p+q},
\qquad (u,v)\mapsto uf+vg.
\]

\begin{block}{Resultant}
\[
  \Res(f,g)=\det S(f,g).
\]
\end{block}
\end{frame}

\begin{frame}{Resultant and GCD}
Assume $R$ is a domain with fraction field $\K$.

\begin{block}{Criterion}
\[
  \Res(f,g)\ne 0
  \quad\Longleftrightarrow\quad
  \gcd(f,g)=1 \text{ in } \K[x].
\]
\end{block}

\vspace{0.5em}
No proof included.
\end{frame}

\section{Subresultants}

\begin{frame}{Subresultant Polynomial}
For $0\le i\le q$ and $0\le j\le i$, let $M_{i,j}(f,g)$ be the corresponding Sylvester-Habicht minor.

\begin{block}{Definition}
\[
  \sResP_i(f,g)=\sum_{j=0}^i \det(M_{i,j}(f,g))x^j.
\]
\end{block}

\begin{block}{Non-defective case}
\[
  \deg\bigl(\sResP_i(f,g)\bigr)=i.
\]
\end{block}
\end{frame}

\begin{frame}{Structure Theorem: Operational Meaning}
\begin{itemize}
  \item Non-zero subresultants encode Euclidean remainders.
  \item Defective subresultants reveal degree drops.
  \item Zero subresultants indicate gaps in the sequence.
\end{itemize}

\vspace{0.8em}
\begin{block}{Consequence}
The GCD and resultant can be computed without division in the fraction field.
\end{block}
\end{frame}

\begin{frame}{Subresultant Algorithm: Skeleton}
\small
\begin{enumerate}
  \item Initialize
  \[
    \sResP_p=f, \qquad \sResP_{p-1}=g.
  \]
  \item Scale the first subresultant:
  \[
    \sResP_q=(-1)^{(p-q)(p-q-1)/2}\lc(g)^{p-q-1}g.
  \]
  \item Repeatedly compute pseudo-remainders:
  \[
    \sResP_{k-1}= -\frac{\Rem(s_k t_{j-1}\sResP_{i-1},\sResP_{j-1})}{s_jt_{i-1}}.
  \]
  \item Fill defective entries when needed.
\end{enumerate}
\end{frame}

\section{Multivariate Polynomials}

\begin{frame}{Multivariate Setting}
Let
\[
  f,g\in \Z[x_1,\dots,x_n].
\]

View them as univariate polynomials in $x_1$:
\[
  f,g\in R[x_1], \qquad R=\Z[x_2,
\dots,x_n].
\]

\begin{block}{Goal}
Compute
\[
  \Res_{x_1}(f,g), \qquad \sResP_i(f,g).
\]
\end{block}
\end{frame}

\begin{frame}{Why Specialize?}
Symbolic computation over
\[
  R=\Z[x_2,\dots,x_n]
\]
can be expensive.

\vspace{0.5em}
Instead, choose evaluation points
\[
  \eta\in\Z^{n-1}
\]
and compute in
\[
  \Z[x_1].
\]

\begin{block}{Workflow}
\[
  \text{evaluate} \quad\to\quad \text{univariate computation} \quad\to\quad \text{interpolate}.
\]
\end{block}
\end{frame}

\section{Proposed Algorithm}

\begin{frame}{Specialization of the Resultant}
Let
\[
  h=\lc_{x_1}(f)\lc_{x_1}(g).
\]

\begin{block}{Valid evaluation point}
If $h(\eta)\ne 0$, then
\[
  \Res_{x_1}(f(\eta),g(\eta))
  =
  \Res_{x_1}(f,g)(\eta).
\]
\end{block}

\begin{alertblock}{Bad points}
Degree drops after evaluation can break the equality.
\end{alertblock}
\end{frame}

\begin{frame}{Degree Bound}
Let
\[
D=\deg_{x_1}(f)\deg_{x_2,\dots,x_n}(g)
+
\deg_{x_1}(g)\deg_{x_2,\dots,x_n}(f).
\]

\begin{block}{Bound}
\[
  \deg\bigl(\Res_{x_1}(f,g)\bigr)\le D.
\]
\end{block}

\begin{block}{Interpolation cost}
For $n=2$, at most $D+1$ evaluation points are needed.
\end{block}
\end{frame}

\begin{frame}{Specialization of Subresultants}
Assume
\[
  \deg_{x_1}(f)>\deg_{x_1}(g)
\]
and $h(\eta)\ne 0$.

\begin{block}{Property}
\[
  \sResP_i(f(\eta),g(\eta))
  =
  \sResP_i(f,g)(\eta)
  \qquad
  0\le i\le \deg_{x_1}(g).
\]
\end{block}

Each coefficient can be reconstructed by interpolation.
\end{frame}

\begin{frame}{Non-defective Subresultants}
Computing only non-defective subresultants can be faster.

\vspace{0.7em}
But evaluation must preserve the degree pattern:
\[
  \deg\bigl(\sResP_i(f(\eta),g(\eta))\bigr)
  =
  \deg\bigl(\sResP_i(f,g)\bigr).
\]

\begin{alertblock}{Extra condition}
Avoid points where a non-defective subresultant becomes defective.
\end{alertblock}
\end{frame}

\begin{frame}{Proposed Bivariate Resultant Algorithm}
For $f,g\in\Z[x_1,x_2]$:
\[
  N=\deg_{x_1}(f)\deg_{x_2}(g)
  +
  \deg_{x_1}(g)\deg_{x_2}(f).
\]

\begin{enumerate}
  \item Choose $N+1$ good evaluation points.
  \item Compute
  \[
    y_i=\Res(f(x_i),g(x_i)).
  \]
  \item Interpolate
  \[
    R=\operatorname{interpolate}(x_i,y_i,N).
  \]
\end{enumerate}
\end{frame}

\begin{frame}{Choosing Evaluation Points}
Implemented strategies:
\begin{itemize}
  \item Random Points
  \item Random Positive Points
  \item Small Random Points
  \item Small Random Positive Points
  \item Ordered Points
  \item Ordered Positive Points
\end{itemize}

\vspace{0.5em}
Small and ordered points usually reduce interpolation cost.
\end{frame}

\section{Conclusion}

\begin{frame}{Summary}
\begin{itemize}
  \item Subresultants control coefficient growth in GCD/resultant computations.
  \item Specialization is valid away from leading-coefficient zeros.
  \item Interpolation recovers multivariate resultants and subresultants.
  \item Evaluation point choice is critical for performance.
  \item Ordered or small-bit-size points work best in the benchmarks.
\end{itemize}
\end{frame}

\begin{frame}{Next Steps}
\begin{itemize}
  \item Extend from resultants to full subresultant sequences.
  \item Improve handling of defective subresultants.
  \item Tune point selection for $\Z$, $\Q$, and $\Fp$.
  \item Compare directly against Wolfram, Maple, and FLINT interfaces.
\end{itemize}
\end{frame}
